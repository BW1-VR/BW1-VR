cmake_minimum_required(VERSION 3.20)
project(bwvr_xr_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
  message(FATAL_ERROR "The bridge targets Windows.")
endif()
option(BWVR_BRIDGE_CHECK_ONLY "Build the x86 transport check without OpenXR" OFF)
if(BWVR_BRIDGE_CHECK_ONLY)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "The transport check needs -A Win32.")
  endif()
  add_executable(bwvr-bridge-check transport_test.cpp client.cpp)
  target_link_libraries(bwvr-bridge-check PRIVATE d3d11 dxgi)
  if(MSVC)
    target_compile_options(bwvr-bridge-check PRIVATE /W4 /permissive-)
  endif()
  return()
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The OpenXR host needs -A x64.")
endif()
set(OPENXR_SDK "" CACHE PATH "OpenXR SDK source folder")
set(OPENXR_BUILD "" CACHE PATH "OpenXR SDK build folder")
set(OPENXR_LOADER "${OPENXR_BUILD}/src/loader/Release/openxr_loader.dll")
if(NOT EXISTS "${OPENXR_LOADER}")
  message(FATAL_ERROR "Build the x64 OpenXR loader first. See README.md.")
endif()
add_executable(bwvr-xr-host host.cpp ../xr_screen.cpp client.cpp)
target_include_directories(bwvr-xr-host PRIVATE "${OPENXR_SDK}/include")
target_link_libraries(bwvr-xr-host PRIVATE
  "${OPENXR_BUILD}/src/loader/Release/openxr_loader.lib" d3d11 dxgi)
if(MSVC)
  target_compile_options(bwvr-xr-host PRIVATE /W4 /permissive-)
endif()
add_custom_command(TARGET bwvr-xr-host POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${OPENXR_LOADER}" "$<TARGET_FILE_DIR:bwvr-xr-host>")
